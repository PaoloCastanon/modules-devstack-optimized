#!/usr/bin/env python3
"""B18 cgroup membership, CPU/PSS and independent control groups, at 1 Hz."""
import argparse
import datetime
import json
import os
from pathlib import Path
import resource
import time

from sample_processes import snapshot


def members(group):
    path = Path('/sys/fs/cgroup') / group.lstrip('/')
    return sorted({int(pid) for entry in path.rglob('cgroup.procs') for pid in entry.read_text().split()})


def collect(groups):
    begin = time.time_ns()
    result = {}
    for label, group in groups.items():
        try:
            pids = members(group)
            values = [snapshot(pid) for pid in pids]
            result[label] = {'processes': values, 'error': None}
        except (OSError, ValueError):
            result[label] = {'processes': [], 'error': 'membership_or_proc_unavailable'}
    return {'utc_ns': begin, 'read_finished_utc_ns': time.time_ns(), 'monotonic_ns': time.monotonic_ns(),
            'groups': result,
            'collector_cpu_s': resource.getrusage(resource.RUSAGE_SELF).ru_utime + resource.getrusage(resource.RUSAGE_SELF).ru_stime}


def identities(row):
    return sorted((p['pid'], p['start_ticks']) for p in row['processes'])


def interval_summary(records, service, start_ns, end_ns, correct_completions_ns, cores=4, ticks=None):
    ticks = ticks or os.sysconf('SC_CLK_TCK')
    inner = [r for r in records if r['utc_ns'] >= start_ns and r['read_finished_utc_ns'] <= end_ns]
    if len(inner) < 3:
        raise ValueError('insufficient samples in common load window')
    first, last = inner[0], inner[-1]
    expected = identities(first['groups'][service])
    if not expected or any(r['groups'][service]['error'] or identities(r['groups'][service]) != expected for r in inner):
        raise ValueError('service membership changed or missing')
    elapsed = (last['utc_ns'] - first['utc_ns']) / 1e9
    if elapsed <= 0 or any(b['utc_ns'] - a['utc_ns'] > 2_500_000_000 for a, b in zip(inner, inner[1:])):
        raise ValueError('resource sampling gap')
    cpu = (sum(p['cpu_ticks'] for p in last['groups'][service]['processes']) - sum(p['cpu_ticks'] for p in first['groups'][service]['processes'])) / ticks
    if cpu < 0:
        raise ValueError('negative CPU delta')
    correct = sum(first['utc_ns'] <= n <= last['utc_ns'] for n in correct_completions_ns)
    pss = [sum(p['pss_bytes'] for p in r['groups'][service]['processes']) for r in inner
           if all(p['pss_bytes'] is not None for p in r['groups'][service]['processes'])]
    if len(pss) != len(inner):
        raise ValueError('PSS required for this campaign')
    overhead = last['collector_cpu_s'] - first['collector_cpu_s']
    return {'window_start_utc_ns': first['utc_ns'], 'window_end_utc_ns': last['utc_ns'],
            'covered_s': elapsed, 'samples': len(inner), 'service_pids_start_ticks': expected,
            'cpu_seconds': cpu, 'cpu_fraction_of_four_cores': cpu / elapsed / cores,
            'correct_requests_common_interval': correct, 'cpu_s_per_correct_request': cpu / correct if correct else None,
            'pss_mean_bytes': sum(pss) / len(pss), 'pss_peak_observed_bytes': max(pss),
            'rss_peak_observed_bytes': max(sum(p['rss_bytes'] for p in r['groups'][service]['processes']) for r in inner),
            'collector_cpu_seconds': overhead, 'collector_core_fraction': overhead / elapsed,
            'max_read_duration_ms': max((r['read_finished_utc_ns'] - r['utc_ns']) / 1e6 for r in inner),
            'rss_note': 'sum may double count shared pages; PSS is primary',
            'alignment_resolution_s': 1}


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--groups', type=Path, required=True)
    parser.add_argument('--output', type=Path, required=True)
    parser.add_argument('--stop-file', type=Path, required=True)
    parser.add_argument('--maximum-seconds', type=int, default=240)
    args = parser.parse_args()
    groups = json.loads(args.groups.read_text())
    if not 10 <= args.maximum_seconds <= 3600 or not groups or any(not v.startswith('/') or '..' in v for v in groups.values()):
        raise ValueError('bounded duration and absolute cgroups required')
    args.output.mkdir(parents=True, exist_ok=False)
    begin, count = time.monotonic(), 0
    with (args.output / 'resources.jsonl').open('x') as stream:
        while time.monotonic() - begin < args.maximum_seconds:
            row = collect(groups)
            stream.write(json.dumps(row, separators=(',', ':')) + '\n')
            stream.flush()
            count += 1
            if args.stop_file.exists():
                break
            time.sleep(max(0, begin + count - time.monotonic()))
    (args.output / 'collector.json').write_text(json.dumps({'samples': count, 'interval_s': 1,
         'clock_ticks_s': os.sysconf('SC_CLK_TCK'), 'timestamp_utc': datetime.datetime.now(datetime.timezone.utc).isoformat(),
         'self_terminated_at_limit': not args.stop_file.exists()}) + '\n')


if __name__ == '__main__':
    main()
